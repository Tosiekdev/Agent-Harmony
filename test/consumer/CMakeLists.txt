cmake_minimum_required(VERSION 3.29)
project(AgentHarmonyConsumer LANGUAGES CXX)
if(TEST_VISUALIZATION)
    find_package(AgentHarmony CONFIG REQUIRED COMPONENTS visualization)
else()
    find_package(AgentHarmony CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE AgentHarmony::core)
if(TEST_VISUALIZATION)
    target_link_libraries(consumer PRIVATE AgentHarmony::visualization)
    target_compile_definitions(consumer PRIVATE TEST_VISUALIZATION)
endif()
enable_testing()
add_test(NAME ConsumerRuns COMMAND consumer)
